{{- if and (not .Values.auth.existingSecret) (not .Values.auth.apiToken) (not .Values.auth.allowUnauthenticated) }}

WARNING: no API token is configured. The worker runs with
TRAFFIC_AI_ENVIRONMENT=production and will refuse to start without one.
Create a Secret and reference it:
  kubectl -n {{ .Release.Namespace }} create secret generic traffic-ai-secrets \
    --from-literal=api-token="$(openssl rand -hex 32)" \
    --from-literal=database-url="postgresql+asyncpg://USER:PASS@HOST:5432/traffic_ai"
  helm upgrade {{ .Release.Name }} <chart> --reuse-values --set auth.existingSecret=traffic-ai-secrets
{{- end }}
